This usage of setTransform was added for cases like specimens. Please mathematically transform setTransform so that it can draw with relative rendering that preserves the previous position (as well as angle and scale), and with image quality as raw drawing.
The center point is (512, 512), but please transform it so that it becomes (0, 0).
Please avoid adding comments.
Transformation example:
```js (before transform)
ctx.lineJoin = "round";
ctx.save();
ctx.lineWidth = 5;
ctx.lineCap = "round";
ctx.beginPath();
ctx.setTransform(-2.715290069580078, -2.71528959274292, 2.71528959274292, -2.715290069580078, 256, 256);
ctx.ellipse(0, 0, 30, 25, 0, 0, 6.283185307179586, false);
ctx.fillStyle = "#FFD363";
ctx.fill();
ctx.save();
ctx.clip();
ctx.fillStyle = "#333333";
ctx.setTransform(-27.15290069580078, -27.152896881103516, 135.7644805908203, -135.76449584960938, 160.96485900878906, 296.7293395996094);
ctx.fillRect(0, 0, 1, 1);
ctx.setTransform(-27.15290069580078, -27.152896881103516, 135.7644805908203, -135.76449584960938, 215.27066040039062, 351.03515625);
ctx.fillRect(0, 0, 1, 1);
ctx.setTransform(-27.15290069580078, -27.152896881103516, 135.7644805908203, -135.76449584960938, 269.57647705078125, 405.3409423828125);
ctx.fillRect(0, 0, 1, 1);
ctx.restore();
ctx.beginPath();
ctx.ellipse(0, 0, 30, 25, 0, 0, 6.283185307179586, false);
ctx.strokeStyle = "#CFAB50";
ctx.stroke();
ctx.strokeStyle = "#333333";
ctx.fillStyle = "#333333";
ctx.lineWidth = 3;
ctx.beginPath();
ctx.moveTo(25, -5);
ctx.quadraticCurveTo(35, -5, 40, -15);
ctx.stroke();
ctx.beginPath();
ctx.arc(40, -15, 5, 0, 6.283185307179586, false);
ctx.fill();
ctx.beginPath();
ctx.moveTo(25, 5);
ctx.quadraticCurveTo(35, 5, 40, 15);
ctx.stroke();
ctx.beginPath();
ctx.arc(40, 15, 5, 0, 6.283185307179586, false);
ctx.fill();
ctx.restore();
```
```js (after transform)
ctx.lineJoin = "round";
ctx.lineCap = "round";

ctx.save();

ctx.lineWidth = 5;

ctx.beginPath();
ctx.ellipse(0, 0, 30, 25, 0, 0, 2 * Math.PI);
```
